cmake_minimum_required(VERSION 3.16)
project(LocalPath LANGUAGES CXX)

add_executable(LocalPath LocalPath/main.cpp LocalPath/third_party/qrcodegen/qrcodegen.cpp)
target_compile_features(LocalPath PRIVATE cxx_std_17)
target_include_directories(LocalPath PRIVATE LocalPath/boost_min)
find_package(Threads REQUIRED)
target_link_libraries(LocalPath PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(LocalPath PRIVATE ws2_32 bcrypt)
endif()
configure_file(LocalPath/index.html ${CMAKE_CURRENT_BINARY_DIR}/index.html COPYONLY)

include(CTest)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:LocalPath>)
  if(UNIX)
    add_test(NAME console_tty COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/console_tty.py $<TARGET_FILE:LocalPath>)
  endif()
endif()
